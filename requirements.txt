# requirements.txt — dependencies for running ADASCRIPT EXAMPLES
# ==============================================================
#
# Three categories of dependency:
#   [nim]    Nim packages  — install with: nimble install <package>
#   [pip]    Python packages — install with: pip install <package>
#   [sys]    System packages — install with your OS package manager
#
# Runtime baseline (not listed below):
#   Nim  >= 2.2.10  (install via choosenim: https://github.com/dom96/choosenim)
#   Zig  >= 0.14.0  (optional: preferred by state_search.ady and others using
#                    --cc:clang --clang.exe:zigcc; py2nim falls back to the
#                    default C compiler when it is absent)
#   Python >= 3.12  (required by py2nim transpiler)
#
# Quick setup:
#   curl -fsSL https://nim-lang.org/choosenim/init.sh | sh  # installs Nim 2.x
#   # Then pin to the tested version:
#   choosenim 2.2.10
#   # Install Zig — optional, only for the exact toolchain the examples were
#   # measured with (no apt package; download from https://ziglang.org/download/):
#   curl -fsSL https://ziglang.org/download/0.14.0/zig-linux-x86_64-0.14.0.tar.xz | tar -xJ -C /opt
#   ln -sf /opt/zig-linux-x86_64-0.14.0/zig /usr/local/bin/zig
#   printf '#!/bin/sh\nexec zig cc "$@"\n' > /usr/local/bin/zigcc && chmod +x /usr/local/bin/zigcc
#   nimble install nimpy db_connector
#   pip install matplotlib
#   apt install fd-find ripgrep bc sqlite3 libpcre3   # adjust for your distro
#   # On Debian/Ubuntu fd-find installs as 'fdfind' — symlink it to 'fd'


# -----------------------------------------------------------------------
# [nim] Nim packages
# -----------------------------------------------------------------------

# nimpy — Python/C interop; required by files that use 'pyimport':
#   primes.ady     (pyimport time)
#   awk_example.ady, test_awk.ady  (nimport re  →  via nimpy)
#   tsp.ady        (pyimport matplotlib)
nimpy >= 0.3.1

# db_connector — SQLite bindings for Nim 2.x; required by:
#   timetable_engine.ady, timetable_backtrack.ady, timetable_sa.ady
db_connector


# -----------------------------------------------------------------------
# [pip] Python packages
# -----------------------------------------------------------------------

# matplotlib — plotting library; required at runtime by:
#   tsp.ady  (pyimport matplotlib.pyplot as plt)
matplotlib

# pygls — Python Language Server Protocol framework; required by:
#   adascript_ls.py  (the Adascript language server)
# Install: python3.13 -m pip install pygls
# Tested version: 2.1.1 (with lsprotocol 2025.0.0)
pygls >= 2.1.1


# -----------------------------------------------------------------------
# [sys] System / external tools
# -----------------------------------------------------------------------

# fd  (fd-find)  — fast file finder; required by:
#   lolcate/lolcate.ady  (shellLines: fd ...)
# Debian/Ubuntu: apt install fd-find
# macOS:         brew install fd
# NOTE: on Debian/Ubuntu the binary is installed as 'fdfind', but the
#       examples invoke it as 'fd'. Add a symlink so the bare name resolves:
#         ln -s "$(command -v fdfind)" /usr/local/bin/fd
fd-find

# rg  (ripgrep)  — fast line searcher; required by:
#   lolcate/lolcate.ady  (shellLines: rg ...)
# Debian/Ubuntu: apt install ripgrep
# macOS:         brew install ripgrep
ripgrep

# bc  — arbitrary-precision calculator; required by:
#   test_expect.ady     (drives bc via PTY)
#   test_shell_block.ady (drives bc via expect/send)
# Debian/Ubuntu: apt install bc
# macOS:         brew install bc
bc

# sqlite3  — SQLite CLI; required for populating the timetable database
#   timetable_backtrack.ady, timetable_sa.ady  (use --db flag)
# Debian/Ubuntu: apt install sqlite3
# macOS:         brew install sqlite
sqlite3

# libpcre3  — PCRE regex runtime; required by test_expect.ady at runtime
# Debian/Ubuntu: apt install libpcre3
# macOS:         included in system libs (no separate install needed)
libpcre3

# zig / zigcc  — Zig C compiler frontend. OPTIONAL: preferred by examples that
# pin the compiler via '#ady2nim-args c --cc:clang --clang.exe:zigcc', but not
# required — py2nim drops a pin whose binary is missing (noting it on stderr)
# and falls back to nim's default C compiler, so these files build without zig:
#   state_search.ady, shortest_path.ady, test_shortest_path.ady,
#   test_state_search.ady, timetable_engine.ady, timetable_sa.ady,
#   timetable_backtrack.ady, TIMETABLE/timetable_server.ady
# 'zigcc' is a thin wrapper around 'zig cc' (a clang frontend); put it on PATH:
#   printf '#!/bin/sh\nexec zig cc "$@"\n' > /usr/local/bin/zigcc && chmod +x /usr/local/bin/zigcc
# Download from: https://ziglang.org/download/  (no OS package available)
# Tested versions: 0.14.0, and 0.16.0 (verified against the full 'make test' run)
zig >= 0.14.0
